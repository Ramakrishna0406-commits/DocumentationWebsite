<script type="module">
/* ============================================================
   CT3 FIREBASE PUBLISH SYSTEM
   ============================================================ */

import {
    doc,
    getDoc,
    setDoc,
    serverTimestamp
} from "https://www.gstatic.com/firebasejs/12.17.1/firebase-firestore.js";

import {
    onAuthStateChanged
} from "https://www.gstatic.com/firebasejs/12.17.1/firebase-auth.js";

import {
    auth,
    db
} from "../js/firebase-config.js";


let ct3CurrentUser = null;


/* ============================================================
   CHECK LOGIN
   ============================================================ */

onAuthStateChanged(
    auth,
    function(user) {

        ct3CurrentUser = user || null;

        const performanceAdmin =
            document.getElementById(
                "performanceBandAdmin"
            );

        const weekAdmin =
            document.getElementById(
                "weekPerformanceAdmin"
            );

        if (
            user &&
            user.email &&
            user.email.toLowerCase() ===
            "rkolluri@gitam.edu"
        ) {

            if (performanceAdmin) {
                performanceAdmin.style.display =
                    "block";
            }

            if (weekAdmin) {
                weekAdmin.style.display =
                    "block";
            }

            console.log(
                "CT3 Admin upload access enabled."
            );

        }
        else {

            if (performanceAdmin) {
                performanceAdmin.style.display =
                    "none";
            }

            if (weekAdmin) {
                weekAdmin.style.display =
                    "none";
            }
        }

        loadPublishedCT3Tables();

    }
);


/* ============================================================
   PUBLISH TABLE
   ============================================================ */

async function publishCT3Excel(type) {

    if (!ct3CurrentUser) {

        alert(
            "Please login as the DATA PULSE administrator."
        );

        return;
    }

    if (
        !ct3CurrentUser.email ||
        ct3CurrentUser.email.toLowerCase() !==
        "rkolluri@gitam.edu"
    ) {

        alert(
            "You are not authorized to publish CT3 tables."
        );

        return;
    }

    const data =
        window.ct3ExcelPreviewData[type];

    if (!data || !data.rows) {

        alert(
            "Please select an Excel file and click Preview first."
        );

        return;
    }

    if (!data.rows.length) {

        alert(
            "The uploaded Excel file contains no data."
        );

        return;
    }

    const confirmed =
        confirm(
            "Publish this table for all DATA PULSE users?"
        );

    if (!confirmed) {
        return;
    }

    try {

        const tableRef =
            doc(
                db,
                "ct3Tables",
                type
            );

        await setDoc(
            tableRef,
            {
                rows: data.rows.map(row => ({ cells: row })),
                fileName: data.fileName,
                sheetName: data.sheetName,
                uploadedBy:
                    ct3CurrentUser.email,
                uploadedAt:
                    serverTimestamp()
            }
        );

        alert(
            "Table published successfully."
        );

        console.log(
            "CT3 table published:",
            type
        );

        renderPublishedCT3Table(
            type,
            data.rows
        );

    }
    catch (error) {

        console.error(
            "CT3 Firebase publish error:",
            error
        );

        alert(
            "Unable to publish the table.\n\n" +
            error.message
        );
    }
}

window.publishCT3Excel = publishCT3Excel;


/* ============================================================
   LOAD PUBLISHED TABLES
   ============================================================ */

async function loadPublishedCT3Tables() {

    await loadPublishedCT3Table(
        "performanceBand"
    );

    await loadPublishedCT3Table(
        "weekPerformance"
    );
}


async function loadPublishedCT3Table(type) {

    try {

        const tableRef =
            doc(
                db,
                "ct3Tables",
                type
            );

        const snapshot =
            await getDoc(tableRef);

        if (!snapshot.exists()) {

            console.log(
                "No published CT3 table:",
                type
            );

            return;
        }

        const data =
            snapshot.data();

        if (
            !data.rows ||
            !Array.isArray(data.rows)
        ) {
            return;
        }

        renderPublishedCT3Table(
            type,
            data.rows
        );

        console.log(
            "Published CT3 table loaded:",
            type
        );

    }
    catch (error) {

        console.error(
            "Unable to load published CT3 table:",
            type,
            error
        );
    }
}


/* ============================================================
   RENDER PUBLISHED TABLE
   ============================================================ */

function renderPublishedCT3Table(type, rows) {

    if (!rows || rows.length === 0) {
        return;
    }

    rows = rows.map(row => Array.isArray(row) ? row : row.cells);

    if (
        type ===
        "performanceBand"
    ) {

        const table =
            document.getElementById(
                "performanceBandTable"
            );

        if (!table) {
            return;
        }

        table.innerHTML = "";

        renderRowsIntoTable(
            table,
            rows
        );

    }
    else if (
        type ===
        "weekPerformance"
    ) {

        const container =
            document.getElementById(
                "weekPerformanceTableContainer"
            );

        if (!container) {
            return;
        }

        container.innerHTML = "";

        const table =
            document.createElement(
                "table"
            );

        table.className =
            "ct3-dynamic-table";

        renderRowsIntoTable(
            table,
            rows
        );

        container.appendChild(
            table
        );
    }
}


/* ============================================================
   GENERIC EXCEL TABLE RENDERER
   ============================================================ */

function renderRowsIntoTable(
    table,
    rows
) {

    const thead =
        document.createElement(
            "thead"
        );

    const tbody =
        document.createElement(
            "tbody"
        );

    const header =
        rows[0] || [];

    const headerRow =
        document.createElement(
            "tr"
        );

    header.forEach(
        function(value) {

            const th =
                document.createElement(
                    "th"
                );

            th.textContent =
                value === null ||
                value === undefined
                    ? ""
                    : String(value);

            headerRow.appendChild(
                th
            );

        }
    );

    thead.appendChild(
        headerRow
    );


    for (
        let i = 1;
        i < rows.length;
        i++
    ) {

        const tr =
            document.createElement(
                "tr"
            );

        const row =
            rows[i] || [];

        for (
            let j = 0;
            j < header.length;
            j++
        ) {

            const td =
                document.createElement(
                    "td"
                );

            const value =
                row[j] === null ||
                row[j] === undefined
                    ? ""
                    : row[j];

            td.textContent =
                String(value);

            tr.appendChild(
                td
            );
        }

        tbody.appendChild(
            tr
        );
    }

    table.appendChild(
        thead
    );

    table.appendChild(
        tbody
    );
}


/* ============================================================
   END CT3 FIREBASE PUBLISH SYSTEM
   ============================================================ */
</script>
</body>
</html>
